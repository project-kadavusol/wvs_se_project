    <script>
        function validateForm() {
            let username = document.getElementById("username").value.trim();
            let password = document.getElementById("password").value.trim();
            let errorDiv = document.getElementById("formError");

            // Clear previous errors
            errorDiv.innerText = "";
            errorDiv.style.display = "none";

            if (username === "" || password === "") {
                errorDiv.innerText = "Both username and password are required!";
                errorDiv.style.display = "block";
                return false;
            }
            return true;
        }

        setTimeout(function() {
            let messageContainer = document.querySelector('.flash-messages');
            if (messageContainer) messageContainer.remove();
        }, 5000);

        // Matrix background effect
        const matrixBackground = document.getElementById('matrixBackground');

        function createMatrixColumn() {
            const column = document.createElement('div');
            column.className = 'matrix-column';
            column.style.left = `${Math.random() * 100}vw`;
            column.style.animationDuration = `${Math.random() * 5 + 2}s`;

            for (let i = 0; i < 20; i++) {
                const bit = document.createElement('span');
                bit.innerText = Math.random() > 0.5 ? '0' : '1';
                bit.style.opacity = Math.random();
                column.appendChild(bit);
            }

            matrixBackground.appendChild(column);

            setTimeout(() => {
                column.remove();
            }, parseFloat(column.style.animationDuration) * 1000);
        }

        setInterval(createMatrixColumn, 100);
    </script>
